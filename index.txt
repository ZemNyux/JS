  <div class="section">
    <h2>Зберегти текст у файл</h2>
    <label for="saveFilePath">Назва файлу (наприклад, example.txt):</label>
    <input type="text" id="saveFilePath" placeholder="example.txt" required>

    <textarea id="content" required></textarea>
    <button onclick="saveFile()">Зберегти</button>

  <div class="section">

    <div id="uploadOutput" class="status-box"></div>
  </div>

  <div class="section">
    <h2>Прочитати файл</h2>
    <label for="loadFilePath">Назва файлу (наприклад, example.txt):</label>
    <input type="text" id="loadFilePath" placeholder="example.txt" required>
    <button onclick="loadFile()">Прочитати</button>
    <div id="loadOutput" class="status-box"></div>
  </div>

  <script>
    async function saveFile() {
      const filePath = document.getElementById('saveFilePath').value;
      const content = document.getElementById('content').value;
      const output = document.getElementById('saveOutput');

      if (!filePath || !content) {
        showStatus(output, 'Помилка: Вкажіть назву файлу та вміст.', true);
        return;
      }

      try {
        const response = await fetch('/save-file', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ filePath, content })
        });
        const result = await response.json();

        if (response.ok) {
          showStatus(output, result.message, false);
        } else {
          showStatus(output, `Помилка: ${result.error}`, true);
        }
      } catch (err) {
        showStatus(output, `Помилка при збереженні файлу: ${err.message}`, true);
      }
    }

    async function uploadFile() {
      const fileInput = document.getElementById('uploadFile');
      const output = document.getElementById('uploadOutput');

      if (!fileInput.files.length) {
        showStatus(output, 'Помилка: Виберіть файл.', true);
        return;
      }

      const formData = new FormData();
      formData.append('file', fileInput.files[0]);

      try {
        const response = await fetch('/upload-file', {
          method: 'POST',
          body: formData
        });
        const result = await response.json();

        if (response.ok) {
          showStatus(output, result.message, false);
        } else {
          showStatus(output, `Помилка: ${result.error}`, true);
        }
      } catch (err) {
        showStatus(output, `Помилка при завантаженні файлу: ${err.message}`, true);
      }
    }

    async function loadFile() {
      const filePath = document.getElementById('loadFilePath').value;
      const output = document.getElementById('loadOutput');

      if (!filePath) {
        showStatus(output, 'Помилка: Вкажіть назву файлу.', true);
        return;
      }

      try {
        const response = await fetch(`/load-file?filePath=${encodeURIComponent(filePath)}`);
        const result = await response.json();

        if (response.ok) {
          showStatus(output, `Вміст файлу "${filePath}":\n\n${result.content}`, false);
        } else {
          showStatus(output, `Помилка: ${result.error}`, true);
        }
      } catch (err) {
        showStatus(output, `Помилка при читанні файлу: ${err.message}`, true);
      }
    }

    function showStatus(element, text, isError) {
      element.textContent = text;
      element.style.display = 'block';
      if (isError) {
        element.classList.add('error');
      } else {
        element.classList.remove('error');
      }
    }
